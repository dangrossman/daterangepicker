Date Range Picker Integrations

Angular

Read the input with ViewChild after the view initializes. The standalone component removes its picker when Angular destroys the view.

App source

report-dates.ts
import {
  AfterViewInit, Component, ElementRef, OnDestroy, ViewChild
} from '@angular/core';
import DateRangePicker from 'daterangepicker';
import 'daterangepicker/daterangepicker.css';

@Component({
  selector: 'report-dates',
  standalone: true,
  template: '<label>Report dates<input #dates type="text"></label>'
})
export class ReportDates implements AfterViewInit, OnDestroy {
  @ViewChild('dates', { static: true }) input!: ElementRef<HTMLInputElement>;
  private picker?: DateRangePicker;

  ngAfterViewInit() {
    this.picker = new DateRangePicker(this.input.nativeElement, { singleCalendar: true });
  }

  ngOnDestroy() {
    this.picker?.remove();
  }
}
main.ts
import '@angular/compiler';
import { provideZonelessChangeDetection } from '@angular/core';
import { bootstrapApplication } from '@angular/platform-browser';
import { ReportDates } from './report-dates';

bootstrapApplication(ReportDates, {
  providers: [provideZonelessChangeDetection()]
}).catch(console.error);
